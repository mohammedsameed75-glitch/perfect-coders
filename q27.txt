units = int(input("Enter units: "))
if units <= 100:
    bill = units * 4
elif units <= 200:
    bill = 100 * 4 + (units - 100) * 5
elif units <= 300:
    bill = 100 * 4 + 100 * 5 + (units - 200) * 6
else:
    bill = 100 * 4 + 100 * 5 + 100 * 6 + (units - 300) * 8
print("Electricity Bill =", bill)

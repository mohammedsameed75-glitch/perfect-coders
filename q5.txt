def power(x, n):
    if n == 0:
        return 1
    return x * power(x, n - 1)
x = float(input("Enter the base: "))
n = int(input("Enter the exponent: "))
if x == 0 and n < 0:
    print("Undefined")
else:
    if n < 0:
        print("Result:", 1 / power(x, -n))
    else:
        print("Result:", power(x, n))
